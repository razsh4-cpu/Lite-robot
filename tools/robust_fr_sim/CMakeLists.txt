cmake_minimum_required(VERSION 3.16)
project(robust_fr_sim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(robust_fr_sim main.cpp)
target_include_directories(robust_fr_sim PRIVATE "${REPO}" "${REPO}/third_party/eigen" "${REPO}/third_party/mujoco/x86/include")
target_link_libraries(robust_fr_sim PRIVATE "${REPO}/third_party/mujoco/x86/lib/libmujoco.so")
set_target_properties(robust_fr_sim PROPERTIES BUILD_RPATH "${REPO}/third_party/mujoco/x86/lib")
enable_testing()
add_test(NAME geometry_and_gates COMMAND robust_fr_sim --self-test)
